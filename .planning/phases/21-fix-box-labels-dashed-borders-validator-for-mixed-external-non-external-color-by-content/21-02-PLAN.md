<objective>
Add validation to prevent mixing external and non-external units in C1 boxes, and color C1 boxes based on their contents (grey for externals, dark blue for non-externals).

Purpose: C1 boxes should be semantically consistent - either all external units or all internal units. This prevents confusing diagrams where a single box contains both external systems and internal components. The box color should reflect its contents: grey boxes contain external units, dark blue boxes contain internal units.

Output: Validation rule and box color determination logic.
</objective>

       // externalTypes are the unit types that represent external boundaries.
       var externalTypes = map[model.UnitType]bool{
           model.TypePersonExternal:  true,
           model.TypeSystemExternal:  true,
           model.TypeDbExternal:      true,
           model.TypeQueueExternal:   true,
       }

       // ValidateBoxMixedContents checks that C1 boxes do not contain both
       // external and non-external units simultaneously.
       // Returns all errors found (not fail-fast).
       func ValidateBoxMixedContents(index map[string]*UnitInfo) ValidationErrors {
           var errors ValidationErrors

           for path, info := range index {
               // Only validate C1 boxes (TypeBox)
               if info.Unit.Type != model.TypeBox {
                   continue
               }

               // Skip boxes without subunits
               if len(info.Unit.Subunits) == 0 {
                   continue
               }

               hasExternal := false
               hasNonExternal := false

               for _, subunit := range info.Unit.Subunits {
                   if externalTypes[subunit.Type] {
                       hasExternal = true
                   } else {
                       hasNonExternal = true
                   }
               }

               if hasExternal && hasNonExternal {
                   errors = append(errors, &ValidationError{
                       Message: fmt.Sprintf(`box "%s" cannot contain both external and non-external units`, path),
                       Path:    path,
                   })
               }
           }

           return errors
       }
       ```

       // hasExternalSubunits checks if a C1 box contains any external units.
       // Used to determine box color based on contents.
       func hasExternalSubunits(unit *model.Unit) bool {
           if unit == nil {
               return false
           }
           for _, subunit := range unit.Subunits {
               if IsExternalType(subunit.Type) {
                   return true
               }
           }
           return false
       }
       ```

    2. In internal/graph/shapes.go, add GetBoxStyleByContents function:

       ```go
       // GetBoxStyleByContents returns the style for C1 boxes based on their contents.
       // - Box with ANY external units: grey border (PersonExternalBorder)
       // - Box with ONLY non-external units: dark blue border (PersonBorder)
       // This is only used for TypeBox (C1 level). C2/C3 boxes use standard level colors.
       func GetBoxStyleByContents(unit *model.Unit) *NodeStyle {
           if hasExternalSubunits(unit) {
               return &NodeStyle{
                   FillColor:   "",
                   BorderColor: model.PersonExternalBorder, // Grey
                   FontColor:   model.PersonExternalBorder,
                   BorderStyle: "dashed",
               }
           }
           return &NodeStyle{
               FillColor:   "",
               BorderColor: model.PersonBorder, // Dark blue
               FontColor:   model.PersonBorder,
               BorderStyle: "dashed",
           }
       }
       ```
