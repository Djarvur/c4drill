           if g == nil {

           defer wasmMutex.Unlock()
           ctx := context.Background()
           gv, err := graphviz.New(ctx)

           if err != nil {

           defer cg.Close()

           var buf bytes.Buffer
           if err := gv.Render(ctx, cg, format, &buf); err != nil {
               return nil, fmt.Errorf("render output: %w", err)
           }
           return buf.Bytes(), nil
       }
