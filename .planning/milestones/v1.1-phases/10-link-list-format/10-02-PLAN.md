```toml

[[unit.link]]

technology = "HTTP"
description = "calls"

# Multiple links to same peer:
[[api.link]]
peer = "db"
technology = "HTTP"
description = "queries data"

[[api.link]]
peer = "db"
technology = "TCP"
description = "health checks"

# Incoming link:
[[db.linkFrom]]

technology = "HTTP"
```

OLD syntax (to replace):
```toml

technology = "HTTP"

# or inline:
link = { "target" = { technology = "HTTP" } }
```

[api]

name = "API Service"

[[api.linkFrom]]

```

```toml
[[api.link]]

technology = "TCP"
description = "Health checks"
```
